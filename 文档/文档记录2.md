2.使用`opencv`检测识别

相机启动节点

```python
import cv2
import rclpy
from sensor_msgs.msg import Image
from rclpy.node import Node
class camera_test(Node):
    def __init__(self):
        super().__init__('FaBu_CheShi')
        self.get_logger().info("摄像头节点已经启动")
        self.cap = cv2.VideoCapture(0)# 创建摄像头对象，0 表示默认摄像头(使用第一个检测到的摄像头)
        self.pub=self.create_publisher(Image,'/camera/image',10)
        self.camera()
    def camera(self):
        if not self.cap.isOpened():
            self.get_logger().info("摄像头打开失败")
            exit()
        while True:
            ret,frame = self.cap.read()#读取摄像头的帧文件，ret表示读取是否成功，frame为numpy数组，包含实际读取到的图像数据
            if not ret:
                break
            cv2.imshow("USB camera",frame)
        	self.img_msg=self.bridge.cv2_to_imgmsg(frame,"bgr8")#frame是一个numpy数组包括height, width, channels，需要将其转化为sensor_msgs/Image格式才可以通过话题发布。
        	self.pub.publish(self.img_msg)
        	self.get_logger().info("图像发布成功")
    def camera_destry(self):
        self.get_logger().info("摄像头关闭节点启动")
        cap.release()#release()方法用于释放与摄像头（或视频文件）相关的资源。
        #cv2.destroyAllWindows()#用于关闭所有由 OpenCV 创建的窗口
        self.get_logger().info("操作完成")
def main():
    rclpy.init(args=None)
    node=camera_test()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        self.get_logger().info("摄像头节点关闭")
    finally:
        node.camera_destry()
        rclpy.shutdown()

```

3.人脸检查节点(`Haar`或`DNN`)

```python
#人脸检测流程(使用opencv的Haar检测人脸)
使用cv2下的CascadeClassifier方法加载Haar级联分类器->订阅相机发布的视频帧信息->使用CascadeClassifier下的detectMultiScale检测人脸位置
```

```python
import traceback
import os
import rclpy
from rclpy.node import Node
import cv2
from cv_bridge import CvBridge
from sensor_msgs.msg import Image
from geometry_msgs.msg import Point
class FaceDetectorHaar(Node):
    def __init__(self):
        super().__init__('Face_Find_shiyan')
        self.get_logger().info('人脸检测节点启动')
        self.bridge = CvBridge()
        cascade_path = '/home/radxa/opencv_haarcascades/haarcascade_frontalface_default.xml'#OpenCV 提供的人脸识别 Haar 级联分类器 文件，用于检测正面人脸。需要cv2.CascadeClassifier()加载
        self.face_cascade = cv2.CascadeClassifier(cascade_path)
        if self.face_cascade.empty():#用于检查是否成功加载了 OpenCV 的 Haar 级联分类器，若加载失败会返回true
            self.get_logger().error(f'加载级联分类器失败: {cascade_path}. 请检查文件是否存在。节点仍会运行，但不会检测人脸。')
        self.sub = self.create_subscription(Image, '/camera/image', self.callback, 10)
        self.pub = self.create_publisher(Point, '/face_center', 10)
        self.get_logger().info('订阅 /camera/image 话题以接收图像')
    def callback(self, msg):
        self.get_logger().debug('收到图像消息，开始处理')
        try:
            if self.bridge is None:
                self.get_logger().warning('收到图像但没有 cv_bridge 来转换图像')
                return
            frame = self.bridge.imgmsg_to_cv2(msg, 'bgr8')#收到视频帧消息，将其转化为numpy格式用于人脸检测
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)#将彩色图片转化为灰度图片，转化后gray是一个二维数组(height weight)
            if self.face_cascade is None or self.face_cascade.empty():
                self.get_logger().warning('未加载有效的级联分类器，跳过检测')
                return
            faces = self.face_cascade.detectMultiScale(
                gray,
                scaleFactor=1.1,
                minNeighbors=5,
                minSize=(60, 60)
            )#detectMultiScale 方法用于检测图像中的目标，返回一个包含检测到的目标（人脸）的位置和大小的矩形框，grey是输入图像，
            #scaleFactor 是图像缩放的比例因子即每次图像处理时图像的大小都会增加10%
            #minNeighbors 是一个阈值参数，它决定了一个候选区域是否被认为是一个真实的人脸。表示在一个候选区域被认为是人脸之前，周围应该有多少个邻近的矩形框。在一个窗口内至少需要 minNeighbors 个相邻矩形框被检测为人脸，该区域才会被确认。
            #minSize 设置检测目标的最小尺寸，单位是像素。这个参数指定了被检测的目标（例如人脸）最小的宽度和高度。
            #face是检测到的人脸的坐标列表。它包含了图像中每个检测到的人脸的矩形框（x, y, w, h）
            h, w = frame.shape[:2]#frame.shape 是 NumPy 数组的一个属性，它返回一个包含数组维度的元组。在图像的情况下，这个元组通常包含图像的高度，宽度，通道数。
            point = Point()
            point.x = -1.0
            point.y = -1.0
            point.z = -1.0

            if len(faces) > 0:#如果没有检测到人脸则face会为空的列表
                (x, y, w_face, h_face) = faces[0]#检测到的第一个人脸
                cx = x + w_face / 2.0
                cy = y + h_face / 2.0
                size = float(w_face * h_face)

                point.x = (cx - w / 2.0) / (w / 2.0)
                point.y = (cy - h / 2.0) / (h / 2.0)
                point.z = size

            self.pub.publish(point)
            self.get_logger().debug('已发布人脸位置信息')
        except Exception as e:
            self.get_logger().error(f'回调处理出错: {e}')
            self.get_logger().debug(traceback.format_exc())

def main():
    rclpy.init()
    node = None
    try:
        node = FaceDetectorHaar()
        rclpy.spin(node)
    except Exception as e:
        # 明确记录异常而不是静默退出
        print('face_find 节点启动/运行时发生异常:', e)
        traceback.print_exc()
    finally:
        if node is not None:
            try:
                node.destroy_node()
            except Exception:
                pass
        rclpy.shutdown()
if __name__ == '__main__':
    main()
```

4.人脸跟随

```python
import rclpy 
from rclpy.node import Node
from geometry_msgs.msg import Twist, Point

def FaceFollower(Node):
    def __init__(self):
        super().__init__("cheshi")
 rospy.Subscriber("/face_center", Point, self.face_callback)
        self.pub = rospy.Publisher("/cmd_vel", Twist, queue_size=1)
    	self.follow_distance = 20000 #期望人脸面积
    def face_callback(self,msg):
        twist = Twist()
        if msg.x == -1:     # 没检测到人脸
            twist.angular.z = 0.0
            twist.linear.x = 0.0
            self.pub.publish(twist)
            return
        twist.angular.z = -msg.x * 0.8
        error = self.follow_distance - msg.z
        twist.linear.x = error * 0.00005  # 比例控制
        # 限速
        twist.linear.x = max(min(twist.linear.x, 0.3), -0.3)
        self.pub.publish(twist)
        
if __name__ == "__main__":
    rospy.init_node("face_follower")
    FaceFollower()
    rospy.spin()
```

5.相机启动人脸识别结合节点

```python
import cv2
import os
import rclpy
from sensor_msgs.msg import Image
from rclpy.node import Node
from cv_bridge import CvBridge
from geometry_msgs.msg import Point
import traceback

class camera(Node):
    def __init__(self):
        super().__init__('FaBu_CheShi')
        self.get_logger().info("熊猫头节点已经启动")
        self.cascade_path='/home/radxa/opencv_haarcascades/haarcascade_frontalface_default.xml'
        if not os.path.exists(self.cascade_path):
            self.get_logger().error("人脸库地址错误")
        self.face_find=cv2.CascadeClassifier(self.cascade_path)
        if self.face_find.empty():
            self.get_logger().error(f'加载级联分类器失败: {cascade_path}')
            exit()
        self.cap = cv2.VideoCapture(2)
        self.bridge = CvBridge()
        self.timer = self.create_timer(0.03,self.camera)
        self.face_pub = self.create_publisher(Point,'/face/location',10)
        
    def camera(self):
        if not self.cap.isOpened():
            self.get_logger().info("摄像头打开失败")
            exit()
        ret,frame = self.cap.read()
        if not ret:
            exit()
        try:
            gray = cv2.cvtColor(frame,cv2.COLOR_BGR2GRAY)
            if self.cascade_path is None:
                self.get_logger().error("人脸库为空")
                return
            self.face = self.face_find.detectMultiScale(
                gray,
                scaleFactor = 1.1,
                minNeighbors = 5,
                minSize=(60,60)
            )
            hight,waight = frame.shape[:2]
            point = Point()
            point.x = -1.0
            point.y = -1.0
            point.z = -1.0
            if len(self.face) > 0:
                (x, y, w_face, h_face) = self.face[0]
                cx = x + w_face / 2.0
                cy = y + h_face / 2.0
                size = float(w_face * h_face)

                point.x = (cx - waight / 2.0) / (waight / 2.0)
                point.y = (cy - hight / 2.0) / (hight / 2.0)
                point.z = size         
            self.face_pub.publish(point)
            self.get_logger().info('已发布人脸位置信息')
        except Exception as e:
            self.get_logger().error(f'回调处理出错: {e}')
            self.get_logger().debug(traceback.format_exc())
   
    def camera_destry(self):
        self.get_logger().info("摄像头关闭节点启动")
        self.cap.release()
        cv2.destroyAllWindows()
        self.get_logger().info("操作完成")
def main():
    rclpy.init(args=None)
    node=camera()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        print("摄像头节点关闭")
        traceback.print_exc()
    finally:
        node.camera_destry()
        if rclpy.ok():
            rclpy.shutdown()
```

```python
        self.get_logger().info("get_parameter客户端启动")
        self.declare_parameter('config_name','test_1')

        self.config_name_value = self.get_parameter('config_name').value
        # create_client signature is (srv_type, srv_name)
        self.client = self.create_client(Interface, 'srv_comulication_test')

        # 定时器定期发送请求（也可改为事件驱动）
        self.timer = self.create_timer(1.0, self.client_callback)


    def client_callback(self):
        # 确保服务可用再发送请求
        if not self.client.wait_for_service(timeout_sec=1.0):
            self.get_logger().info('等待服务端连接')
            return

        request = Interface.Request()
        # srv request field is `config_name`
        request.config_name = self.config_name_value

        # 非阻塞调用，使用回调处理响应
        self.future = self.client.call_async(request)
        self.future.add_done_callback(self._on_response)
        #当future结束后会调用self._on_response回调函数（无论是成功还是失败）


    def _on_response(self, future):
        try:
            response = future.result()
            self.get_logger().info(f'Response: {response}')
        except Exception as e:
            self.get_logger().error(f'调用失败: {e}')
```

# `ROS2`生命周期

## `python`需要加入的头文件

```python
import rclpy
from rclpy.lifecycle import LifecycleNode
from rclpy.lifecycle import State
from rclpy.lifecycle import TransitionCallbackReturn
from std_msgs.msg import String
```

## 几个主要状态

```python
unconfigured			节点刚创建，还没初始化
inactive				已初始化，但不工作
active					正在正常工作
finalized				节点结束，不能再用
```

## 次要状态

```python
onConfigure()			加载配置、冗长的设置步骤、获取节点生命周期内使用的资源，如常量发布者/侦听器、内存缓冲区分配等…
onCleanup()				放弃资源，擦除内存。 新的开始，清理状态。
onActivate()			获取传感器等短期资源，激活所有资源。 设置时间短。 启动主节点任务。
onDeactivate()			反向Activatingsteps（反激活）。 暂停执行，释放短期资源。
onShutdown()			最后步骤。 删除剩下的资源等。 没有信息从此处回来。
onError()				错误处理状态。 如果错误可以处理，则恢复到 Unconfigured，否则，转到 Finalized 销毁节点。
```

## 过度状态

```python
configure				初始化资源
activate				开始工作
deactivate				暂停工作
cleanup					释放资源
shutdown				彻底退出
```

## 状态流转图

```python
unconfigured--(configure)-->inactive--(activate)-->active--(deactivate)-->inactive--(cleanup)-->unconfigured--(shutdown)-->finalized
```

## 生命周期接口

```python
服务：
/change_state - 调用触发合法转换
/get_available_transitions - 显示合法的转换
/get_state - 显示当前状态
/get_available_states - 列出所有状态
/get_transition_graph - 显示完整状态机
主题：
/<node_name>__transition_event - 发布正在进行的转换
```

## 生命周期命令行接口

```python
ros2 lifecycle nodes - 列出所有 LC 节点
ros2 lifecycle get - 列出指定节点或所有 LC 节点的当前状态
ros2 lifecycle list - 列出指定节点可能的下一个状态和相应的转换调用（名称和 ID）
ros2 lifecycle set - 在 LC 节点上触发转换（按名称或 ID）。
```

`python`编写生命周期函数时不能继承`Node`类而应该继承`LifecycleNode`类型

生命周期节点依靠回调函数控制行为。

## `on_configure`

初始化参数，创建publisher/timer，但不要开始执行

```python
def on_configure(self, state: State):
    self.get_logger().info('Configuring...')

    self.publisher_ = self.create_lifecycle_publisher(
        String,
        'lifecycle_chatter',
        10
    )
    self.timer_ = self.create_timer(1.0, self.timer_callback)
    self.timer_.cancel()   # ⚠️ 配置阶段不要运行
    return TransitionCallbackReturn.SUCCESS
```

在`on_configure`中应该只有读取参数，创建发布/订阅者，分配资源，不应该发布话题，开启硬件

## `on_activate`(开始干活)

```python
def on_activate(self, state: State):
    self.get_logger().info('Activating...')

    self.publisher_.on_activate()
    self.timer_.reset()

    return TransitionCallbackReturn.SUCCESS
```

这里允许发布数据，启动定时器，启动设备

## `on_deactivate`(暂停)

```python
def on_deactivate(self, state: State):
    self.get_logger().info('Deactivating...')

    self.timer_.cancel()
    self.publisher_.on_deactivate()

    return TransitionCallbackReturn.SUCCESS
```

## `on_cleanup`(释放资源)

```python
def on_cleanup(self, state: State):
    self.get_logger().info('Cleaning up...')

    self.destroy_timer(self.timer_)
    self.destroy_publisher(self.publisher_)

    return TransitionCallbackReturn.SUCCESS
```

`on_shutdown`(退出)

```python
def on_shutdown(self, state: State):
    self.get_logger().info('Shutting down...')
    return TransitionCallbackReturn.SUCCESS
```

```python
node_names				被管理的生命周期节点列表
autostart				是否自动 configure + activate
bond_timeout			节点心跳超时
```

# `ROS2`中的几个延时函数

`timer.sleep(1)`	会阻塞节点，节点无法收消息进行回调

`rclpy.sleep(1)`	会阻塞节点，节点无法收消息进行回调

`await rclpy.sleep(1)`	不会阻塞节点，节点可以正常接收消息回调

`await rclpy.sleep(1)`

节点在等待期间继续运行事件循环，该收消息收消息，该执行回调执行回调，延时完场后再继续执行代码。

**要求**

`await`只能用在异步函数中

代码示例

```python
import rclpy
from rclpy.node import Node

class MyNode(Node):
    def __init__(self):
        super().__init__("test_node")

    # 👉 必须加 async
    async def run_loop(self):
        while rclpy.ok():
            self.get_logger().info("我在运行，节点不卡顿！")
            
            # ✅ 异步非阻塞延时
            await rclpy.sleep(0.5) 

def main(args=None):
    rclpy.init(args=args)
    node = MyNode()
    
    # 运行异步循环
    rclpy.spin(node)
    # 或用：asyncio.run(node.run_loop())

    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

# '分水岭分割示例'
```python
import cv2
import numpy as np

# 1. 读取图像
img = cv2.imread('coins.jpg')
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# 2. 预处理：去噪 + 二值化
ret, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)

# 3. 形态学操作：去除小噪点，平滑边缘
kernel = np.ones((3,3), np.uint8)
opening = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, kernel, iterations=2)

# 4. 确定背景（sure_bg）：膨胀操作，让前景变胖，剩下的黑色区域就是背景
sure_bg = cv2.dilate(opening, kernel, iterations=3)

# 5. 确定前景（sure_fg）：距离变换，提取对象的“核心”区域
dist_transform = cv2.distanceTransform(opening, cv2.DIST_L2, 5)
ret, sure_fg = cv2.threshold(dist_transform, 0.7*dist_transform.max(), 255, 0)
sure_fg = np.uint8(sure_fg)

# 6. 未知区域（unknown）：背景 - 前景
unknown = cv2.subtract(sure_bg, sure_fg)

# 7. 创建标记（Marker）
# 对确定的前景区域进行连通域标记
ret, markers = cv2.connectedComponents(sure_fg)
# 将所有标签+1，让背景区域（原本为0）变为1
markers = markers + 1
# 将未知区域设置为0，这是算法需要求解的地方
markers[unknown == 255] = 0

# 8. 应用分水岭算法！关键一步
markers = cv2.watershed(img, markers)

# 9. 可视化结果：将边界（-1）标记为红色
img[markers == -1] = [255, 0, 0]

cv2.imshow('Result', img)
cv2.waitKey(0)
cv2.destroyAllWindows()
```

# '区域生长示例'
```python

import cv2
import numpy as np

def region_growing(img, seed, threshold):
    """
    区域生长算法的核心实现

    Args:
        img: 输入的灰度图像 (单通道)
        seed: 种子点坐标 (row, col)
        threshold: 灰度差阈值，控制生长严格程度

    Returns:
        region_mask: 一个与图像大小相同的二值掩码，生长区域为True
    """
    # 获取图像尺寸
    height, width = img.shape
    
    # 记录像素是否已被访问，避免重复处理
    visited = np.zeros_like(img, dtype=np.bool_)
    # 创建掩码，用于存储最终生长结果
    region_mask = np.zeros_like(img, dtype=np.bool_)
    
    # 使用栈来存储待处理的像素点 (后进先出)
    stack = [seed]
    # 标记种子点为已访问，并加入区域
    visited[seed] = True
    region_mask[seed] = True

    while stack:
        # 取出一个像素点
        x, y = stack.pop()
        
        # 检查该像素点的四邻域 (上下左右)
        # 如果检查八邻域，可以加上 (dx, dy) 组合为 [(-1,-1), (-1,0), ...]
        for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nx, ny = x + dx, y + dy
            
            # 确保邻域像素在图像范围内
            if 0 <= nx < height and 0 <= ny < width:
                # 如果该邻域像素未被访问
                if not visited[nx, ny]:
                    # 计算该像素与当前处理像素的灰度差，并判断是否小于阈值
                    # 也可以计算与种子点的差值，或与区域平均灰度值的差值
                    if abs(int(img[nx, ny]) - int(img[x, y])) < threshold:
                        # 满足条件，将其标记为区域一部分，并压入栈中以供后续扩展
                        visited[nx, ny] = True
                        region_mask[nx, ny] = True
                        stack.append((nx, ny))
                    else:
                        # 不满足条件，只标记为已访问，避免重复判断
                        visited[nx, ny] = True
    return region_mask

# --- 使用示例 ---
# 1. 以灰度模式读取图像
img = cv2.imread('your_image.jpg', 0)

# 2. 选择一个合适的种子点 (例如，你想分割的区域中心)
seed_point = (100, 100) 

# 3. 设定一个合理的阈值，需要根据图像特性调整
similarity_threshold = 15 

# 4. 执行区域生长
region = region_growing(img, seed_point, similarity_threshold)

# 5. 可视化结果 (可选)
# 将原图与生长区域的掩码叠加，显示分割效果
result_img = cv2.cvtColor(img, cv2.COLOR_GRAY2BGR)
result_img[region] = [0, 0, 255] # 将分割区域标记为红色

cv2.imshow('Original', img)
cv2.imshow('Region Growing Result', result_img)
cv2.waitKey(0)
cv2.destroyAllWindows()
```
