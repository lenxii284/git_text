## 使用adb远程推送文件

### 完整推送流程

```python
# 步骤1：连接到远程设备
adb connect 192.168.1.100:5555
# 步骤2：验证连接
adb devices
# 步骤3：推送文件
adb push document.pdf /sdcard/Download/
# 步骤4：验证文件
adb shell ls -l /sdcard/Download/document.pdf
```

### 多设备管理

```python
# 查看所有设备
adb devices
# 推送到指定设备
adb -s 192.168.1.100:5555 push file.txt /sdcard/
# 或者设置默认设备
export ANDROID_SERIAL=192.168.1.100:5555
adb push file.txt /sdcard/
# 进入设置界面
adb shell am start -a android.settings.SETTINGS
```

### 示例

```python
# 推送APK并安装
adb connect 192.168.1.100:5555
adb push app.apk /data/local/tmp/
adb shell pm install /data/local/tmp/app.apk
# 推送配置文件
adb push config.json /sdcard/myapp/
# 推送整个项目文件夹
adb push project_folder/ /sdcard/myproject/
```

### adb其他命令

```python
# 安装应用
adb install app.apk
adb install -r app.apk  # 覆盖安装
# 卸载应用
adb uninstall com.example.app
# 查看已安装应用
adb shell pm list packages
adb shell pm list packages -f  # 显示apk路径
adb shell pm list packages -s  # 系统应用
adb shell pm list packages -3  # 第三方应用
# 清除应用数据
adb shell pm clear com.example.app
# 推送文件到设备
adb push local.txt /sdcard/
# 从设备拉取文件
adb pull /sdcard/file.txt ./
# 查看设备文件
adb shell ls /sdcard/
# 查看实时日志
adb logcat
# 查看带时间和标签的日志
adb logcat -v time
# 按标签过滤日志
adb logcat -s TAG_NAME
# 按级别过滤日志
adb logcat *:E  # 只显示错误日志
# 清除日志缓冲区
adb logcat -c
adb bugreport > bugreport.zip
adb bugreport: 生成包含设备全面信息的诊断报告
> bugreport.zip: 将输出重定向到zip文件（实际会自动压缩）
```


```python

scp -r ./my_folder alice@192.168.1.100:/home/alice/

scp -C ./large_file.iso alice@192.168.1.100:/home/alice/
```

## launch(python)

```python
#!/usr/bin/env python3

from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, ExecuteProcess, IncludeLaunchDescription, RegisterEventHandler, LogInfo
from launch.conditions import IfCondition, UnlessCondition
from launch.event_handlers import OnProcessStart, OnProcessExit
from launch.substitutions import LaunchConfiguration, PathJoinSubstitution, PythonExpression
from launch_ros.actions import Node, SetParameter
from launch_ros.substitutions import FindPackageShare
from launch.launch_description_sources import PythonLaunchDescriptionSource

def generate_launch_description():
    
    # 启动参数
    use_sim_time = LaunchConfiguration('use_sim_time')
    node_name = LaunchConfiguration('node_name')
    enable_debug = LaunchConfiguration('enable_debug')
    
    return LaunchDescription([
        # 全局参数设置
        SetParameter(name='use_sim_time', value=use_sim_time),
        
        # 参数声明
        DeclareLaunchArgument(
            'use_sim_time',
            default_value='false',
            description='使用仿真时间'
        ),
        
        DeclareLaunchArgument(
            'node_name',
            default_value='my_node',
            description='节点名称'
        ),
        
        DeclareLaunchArgument(
            'enable_debug',
            default_value='false',
            description='启用调试模式'
        ),
        
        # 条件启动的节点
        Node(
            package='demo_nodes_cpp',
            executable='talker',
            name=node_name,
            condition=UnlessCondition(enable_debug),
            parameters=[{
                'use_sim_time': use_sim_time,
            }],
            output='screen',
            emulate_tty=True,
        ),
        
        # 调试节点（仅在启用调试时启动）
        Node(
            package='demo_nodes_cpp',
            executable='talker',
            name=[node_name, '_debug'],
            condition=IfCondition(enable_debug),
            parameters=[{
                'use_sim_time': use_sim_time,
            }],
            prefix='xterm -e gdb --args',
            output='screen',
        ),
        
        # 监听器节点
        Node(
            package='demo_nodes_cpp',
            executable='listener',
            name='listener_node',
            parameters=[{
                'use_sim_time': use_sim_time,
            }],
            remappings=[
                ('chatter', 'my_chatter_topic')
            ],
            output='screen',
            emulate_tty=True,
        ),
        
        # 包含其他launch文件
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource([
                PathJoinSubstitution([
                    FindPackageShare('demo_launch'),
                    'launch',
                    'included_launch.py'
                ])
            ]),
            launch_arguments={
                'use_sim_time': use_sim_time,
                'parent_arg': 'from_parent'
            }.items()
        ),
        
        # 带事件处理的节点
        Node(
            package='demo_nodes_cpp',
            executable='minimal_publisher',
            name='event_publisher',
            output='screen',
        ),
        
        # 事件处理 - 节点启动时
        RegisterEventHandler(
            OnProcessStart(
                target_action=launch_description.get_entity_by_name('event_publisher'),
                on_start=[
                    LogInfo(msg='Publisher node started successfully')
                ]
            )
        ),
        
        # 事件处理 - 节点退出时
        RegisterEventHandler(
            OnProcessExit(
                target_action=launch_description.get_entity_by_name('event_publisher'),
                on_exit=[
                    LogInfo(msg='Publisher node has exited')
                ]
            )
        )
    ])
```

## ROS域ID

```python
export ROS_DOMAIN_ID=??
#命令用于设置ROS2的域ID（Domain ID）
?? 需要替换为具体的数字（通常是0-101）
#作用
1.同一网络中不同域ID的ROS2节点无法通信
2.相同域ID的ROS2节点可以自动发现和通信
3.实现多机器人系统或测试环境的网络隔离
#实际应用示例
# 终端1 - 机器人控制站
export ROS_DOMAIN_ID=10
ros2 launch robot_bringup launch.py
# 终端2 - 单独的分析工具
export ROS_DOMAIN_ID=20
ros2 topic list  # 看不到域ID 10的话题
#开发测试隔离
# 终端A - 运行实际系统
export ROS_DOMAIN_ID=1
ros2 run my_package node1
# 终端B - 测试节点（不会干扰实际系统）
export ROS_DOMAIN_ID=2  
ros2 run test_package test_node
#补充
# 临时设置（当前终端有效）
export ROS_DOMAIN_ID=25
# 永久设置（添加到 ~/.bashrc）
echo "export ROS_DOMAIN_ID=25" >> ~/.bashrc
source ~/.bashrc
# 检查当前域ID
echo $ROS_DOMAIN_ID
# 查看ROS2环境
ros2 doctor
注意：所用使用相同域ID的设备必须在同一网络下。
```

# 启动停止自启服务

```python
sudo systemctl enable ros_launch.service
#功能：启用服务的开机自启动。
#详细解释：这个命令并不会立即启动服务。它只是在文件系统中创建一些符号链接，告诉 systemd：“当系统启动到多用户模式时，请自动运行这个服务”。执行后，每次服务器重启，ros_launch.service 所代表的 ROS 启动文件都会自动运行。
#使用场景：当你确认某个 ROS 节点或启动文件需要随系统一直运行
sudo systemctl disable ros_launch.service
#功能：禁用服务的开机自启动。
#详细解释：这个命令与 enable 相反。它不会停止当前正在运行的服务，只会移除之前由 enable 创建的符号链接。执行后，系统重启时，这个服务将不会再自动启动。
#使用场景：当你不再需要某个 ROS 服务随系统自动启动时。
sudo systemctl start ros_launch.service
#功能：立即启动服务。
#详细解释：这个命令会立即尝试运行 ros_launch.service 中定义的命令（例如 roslaunch your_pkg your_launch.launch）。它只影响服务的当前状态，不改变其开机自启动的设置。
#使用场景：在调试、手动控制或服务被停止后，你需要立即运行它。
sudo systemctl stop ros_launch.service
#功能：立即停止服务。
#详细解释：这个命令会向服务发送停止信号，请求它优雅地关闭。它只影响服务的当前状态，不改变其开机自启动的设置。
#使用场景：当你需要手动停止一个正在运行的 ROS 服务时。
sudo systemctl status ros_launch.service
#功能：查看服务的状态信息。这是最常用的诊断命令之一。
#详细解释：它会显示服务的丰富信息，包括：
#运行状态：active (running), inactive (dead), failed, activating 等。
#是否启用开机自启动：enabled 或 disabled。
#最近的日志片段：显示服务最近输出的几条日志，对于快速排查问题非常有用。
#进程ID（PID）：服务主进程的 ID。
#资源占用：（有时会显示）内存和 CPU 的使用情况。
#使用场景：任何时候你想确认服务是否在运行、为什么启动失败等。
sudo journalctl -u ros_launch.service -f
#您正在使用 journalctl 命令来实时查看 ROS 启动服务的日志。这个命令会：
#-u ros_launch.service：指定查看特定 systemd 服务的日志
#-f：实时跟踪日志输出（类似 tail -f）
#常见使用场景：
#监控 ROS 节点的启动状态
#试服务启动失败的原因
#看 ROS 节点的实时输出信息
其他有用的journalctl命令
# 查看最近的服务日志（不实时跟踪）
sudo journalctl -u ros_launch.service
# 查看指定时间段的日志
sudo journalctl -u ros_launch.service --since "2024-01-01 00:00:00"
# 查看最近1小时的日志
sudo journalctl -u ros_launch.service --since "1 hour ago"
# 查看日志并显示所有字段（包括进程ID等）
sudo journalctl -u ros_launch.service -o verbose
# 查看错误级别的日志
sudo journalctl -u ros_launch.service -p err
```

# OTA是什么

```python
OTA 指的是通过无线网络（如 Wi-Fi 或蜂窝移动网络）来传输数据、配置设备或安装软件更新，而无需使用物理线缆（如 USB 数据线）连接。
在 Android 系统和智能手机的语境下，OTA 几乎特指“系统软件更新”。
简单来说，OTA 就是一种“无线升级”技术。
在手机领域，它特指通过 Wi-Fi 或移动网络来下载和安装系统更新。您之前遇到的 adb bugreport 命令生成的报告，以及 ro.build.version.incremental 属性，都是开发和测试人员用来分析和验证 OTA 更新过程是否正常、排查更新后问题的重要工具。
```

# USB串口绑定

1.查看当前连接的USB设备

```
lsusb
```

2.查看详细的设备信息

```python
udevadm info --name=/dev/ttyUSB0 --attribute-walk
# 或者使用更简洁的方式查看关键属性
udevadm info --name=/dev/ttyUSB0 --query=property
```

3.创建udev规则文件

```
sudo nano /etc/udev/rules.d/99-right-arm.rules
# Bind ttyUSB0 to right_arm
KERNEL=="ttyUSB0", SYMLINK+="right_arm", MODE="0666"
```

4.重新加载并应用规则

```python
# 重新加载udev规则
sudo udevadm control --reload-rules
sudo udevadm trigger
# 验证是否创建成功
ls -l /dev/right_arm
```

使用VID/PID跟可靠

1.获取设备的VID和PID

```python
# 查看设备的VID和PID
udevadm info --name=/dev/ttyUSB0 | grep -E "(ID_VENDOR_ID|ID_MODEL_ID)"
```

输出类似

```
E: ID_VENDOR_ID=10c4
E: ID_MODEL_ID=ea60
```

2.创建基于VID/PID额规则

```python
sudo nano /etc/udev/rules.d/99-right-arm.rules
#添加内容
# Bind specific USB device by VID/PID to right_arm
SUBSYSTEM=="tty", ATTRS{idVendor}=="10c4", ATTRS{idProduct}=="ea60", SYMLINK+="right_arm", MODE="0666"
```

# TCP通信代码

### TCP通信流程

```python
通信模型
ROS2 节点A（TCP服务器） ⇄ TCP Socket ⇄ 设备/PC端（TCP客户端）
```

```python
#服务端
1.创建socket  socket(AF_INET, SOCK_STREAM) 创建 TCP 套接字对象
2.绑定地址     bind(('IP', port))			指定本地监听的IP和端口
3.开始监听     listen()						开始等待连接
4.建立连接     accept()						三次握手成功后形成连接
5.数据通信     send() / recv()				双方进行收发数据
6.关闭连接     close()						四次挥手断开连接
#客户端
1.创建socket  同上
2.建立连接     connect()
3.数据通信     send() / recv()
4.关闭连接     close()
```

### 服务端

逻辑流程

```python
初始化节点 → 启动TCP服务器线程 → 等待客户端连接
           ↓
定时读取JSON文件 → 生成消息 → 发送数据
```

建立TCP服务器

```python
self.TCP_serve = socket.socket(socket.AF_INET, socket.SOCK_STREAM)   #创建socket
self.TCP_serve.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)	#设置套接字在关闭后可以立即重用相同的地址和端口。避免"Address already in use"错误
self.TCP_serve.bind(('0.0.0.0', 8888))	#绑定地址
self.TCP_serve.listen(5)	#开始监听
#socket.AF_INET → IPv4 地址族
#socket.SOCK_STREAM → TCP 协议（面向连接、可靠传输）
#SO_REUSEADDR → 允许服务器重启后立即重用端口
#'0.0.0.0' → 监听本机所有网络接口
#listen(5) → 最多5个等待队列的客户端连接
```

接受客户端连接

```python
self.conn, self.add = self.TCP_serve.accept()
#阻塞等待客户端连接
#self.conn：专属客户端连接对象
#self.add：客户端地址 (ip, port)
```

数据通信

```python
self.message="123abc"
if hasattr(self,'conn'):	#检查对象是否有 conn 属性避免在 conn 不存在时访问它导致 AttributeError
       self.conn.send(self.message.encode('utf-8'))	#通过连接对象发送编码后的数据，将字符串转换为UTF-8编码的字节数据TCP套接字只能发送字节数据，不能直接发送字符串
```

关闭连接

```python
self.conn.shutdown(socket.SHUT_WR)	#停止读写
time.sleep(1)
self.conn.close()	#完全关闭连接
```

### 客户端

逻辑流程

```python
初始化节点 → 创建Socket → 连接服务器 → 循环接收消息
```

建立连接

```python
self.serve_address=('192.168.113.72',8888)
self.TCP_client.connect(self.serve_address)
```

接收消息并解码

```python
self.data = self.TCP_client.recv(1024)	#从TCP连接接收数据，1024表示最大接收字节数，返回空字节串b''表示连接关闭
self.message = self.data.decode('utf-8')	#将字节数据解码为字符串'utf-8' 指定编码格式
self.get_logger().info(f"收到消息: {self.message}")	
```

```python
#导入模块
import cv2          # OpenCV，用于摄像头操作和图像显示
import rclpy        # ROS2 Python 客户端库
from rclpy.node import Node  # ROS2 节点基类
```

```python
#节点类定义
class camera_test(Node):
    def __init__(self):
        super().__init__('FaBu_CheShi')  # 初始化父类，节点名为'FaBu_CheShi'
        self.get_logger().info("摄像头节点已经启动")  # ROS2 日志输出
        self.cap = cv2.VideoCapture(0)   # 创建摄像头对象，0 表示默认摄像头(使用第一个检测到的摄像头)
```

```python
#功能实现
 def run_camera(self):
        #检测摄像头是否打开
        if not self.cap.isOpened():
            self.get_logger().info("摄像头打开失败")
            return False
        while rclpy.ok():  # 使用 ROS2 运行状态作为循环条件
            ret, frame = self.cap.read()#用于从摄像头或视频文件读取帧
            if not ret:#ret表示读取是否成功
                self.get_logger().info("读取摄像头帧失败")
                break
            cv2.imshow("USB camera", frame)#frame:numpy数组，包含实际读取到的图像数据
            if cv2.waitKey(1) & 0xFF == ord('q'):#检测键盘输入，按'q'键退出
                break
        self.cleanup()
        return True
    def cleanup(self):
        if hasattr(self, 'cap'):
            self.cap.release()
        cv2.destroyAllWindows()
```

```python
def main():
    rclpy.init()
    node = camera_test()  # 创建节点实例
    try:
        node.run_camera()  # 运行摄像头功能
    except KeyboardInterrupt:
        pass
    finally:
        node.cleanup()  # 确保资源被释放
        rclpy.shutdown()
```

# `python`中的`callback_group`

`ROS2 Python`有两种回调组

1.`MutuallyExclusive（互斥）`同一回调组中回调同一时间只能跑一个

2.`Reentrant（可重入）`同一组内回调可以并行执行。

需要的依赖

```python
import rclpy
from rclpy.node import Node
from rclpy.callback_groups import MutuallyExclusiveCallbackGroup, ReentrantCallbackGroup
from rclpy.executors import MultiThreadedExecutor  # 多线程执行器必须配合使用
from std_msgs.msg import String
```

3.创建回调组

```python
# 1) 互斥回调组（常用，防死锁）
self.cb_group_mutex = MutuallyExclusiveCallbackGroup()
# 2) 可重入回调组
self.cb_group_reent = ReentrantCallbackGroup()
```

4.绑定到订阅 / 定时器 / 服务（指定 callback_group 参数）

```python
# 订阅话题，绑定互斥回调组
self.sub = self.create_subscription(
    String,
    "/dock_state",
    self.dock_state_callback,
    10,
    callback_group=self.cb_group_mutex  # 关键：指定回调组
)

# 定时器，绑定可重入回调组
self.timer = self.create_timer(
    1.0,
    self.timer_callback,
    callback_group=self.cb_group_reent
)

# 服务客户端，绑定回调组
self.client = self.create_client(
    Empty,
    "/cmd_dock",
    callback_group=self.cb_group_mutex
)
```

5.多线程执行器

```python
def main(args=None):
    rclpy.init(args=args)
    node = YourNodeName()
    # 用 MultiThreadedExecutor 替代默认单线程 spin
    executor = MultiThreadedExecutor()
    executor.add_node(node)
    try:
        executor.spin()
    except KeyboardInterrupt:
        pass
    node.destroy_node()
    rclpy.shutdown()
```
